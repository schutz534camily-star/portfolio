    lock() { if (cb) cb.hidden = true; },

    done() { t.remove(); },
    fail(msg) { t.className = 'pf-toast bad'; t.textContent = msg; setTimeout(() => t.remove(), 7000); }
  };
}

class ApiError extends Error { constructor(m, status, data) { super(m); this.status = status; this.data = data; } }
async function api(method, path, body) {
  const r = await fetch(path, {
    method, credentials: 'same-origin',
    headers: { 'x-requested-with': 'portfolio', ...(body !== undefined ? { 'content-type': 'application/json' } : {}) },
    body: body !== undefined ? JSON.stringify(body) : undefined
  });
  const data = await r.json().catch(() => ({}));
  if (r.status === 401) { toast('Sua sessão acabou. Entre de novo.', 'bad', 0); setTimeout(() => location.href = '/admin', 1500); }
  if (!r.ok) throw new ApiError(data.error || 'Erro no servidor', r.status, data);
  return data;
}
let queue = Promise.resolve();
function save() {
  queue = queue.then(async () => {
    try {
      const r = await api('PUT', '/api/content', { clients: PF.clients, rev: PF.rev });
      PF.rev = r.rev; toast('Salvo ✓', 'ok', 1800);
    } catch (e) {
      if (e.status === 409) { toast(e.message, 'bad', 0); setTimeout(() => location.reload(), 2500); }
      else { toast(e.message, 'bad', 6000); await reload(); }
      throw e;
    }
  });
  return queue.catch(() => {});
}

}

const RATIOS = [['9/16', 'Vertical (9:16)'], ['16/9', 'Horizontal (16:9)'], ['4/5', 'Retrato (4:5)'], ['1/1', 'Quadrado (1:1)'], ['4/3', 'Horizontal clássico (4:3)'], ['3/4', 'Retrato (3:4)']];
function nearestRatio(w, h) {
  const q = w / h; let best = '1/1', d = 9;
  for (const [r] of RATIOS) { const [a, b] = r.split('/').map(Number); const x = Math.abs(Math.log(q / (a / b))); if (x < d) { d = x; best = r; } }
  return best;
}
/* GIF: garante que roda em loop infinito (acrescenta/ajusta o bloco NETSCAPE2.0) */
async function gifLoop(file) {
  const b = new Uint8Array(await file.arrayBuffer());
  if (b.length < 14 || String.fromCharCode(...b.slice(0, 3)) !== 'GIF') return file;
  let p = 13; if (b[10] & 0x80) p += 3 * (1 << ((b[10] & 7) + 1));
  for (let q = p; q + 18 < b.length && b[q] === 0x21;) {
    if (b[q + 1] === 0xFF && b[q + 2] === 0x0B && String.fromCharCode(...b.slice(q + 3, q + 14)) === 'NETSCAPE2.0') {
      b[q + 16] = 0; b[q + 17] = 0; return new File([b], file.name, { type: 'image/gif' });
    }
    if (b[q + 1] === 0xF9 || b[q + 1] === 0xFE || b[q + 1] === 0x01 || b[q + 1] === 0xFF) { q += 2; while (q < b.length && b[q] !== 0) q += b[q] + 1; q++; } else break;
  }
  const ext = new Uint8Array([0x21, 0xFF, 0x0B, ...'NETSCAPE2.0'.split('').map(c => c.charCodeAt(0)), 0x03, 0x01, 0x00, 0x00, 0x00]);
  const out = new Uint8Array(b.length + ext.length); out.set(b.slice(0, p)); out.set(ext, p); out.set(b.slice(p), p + ext.length);
  return new File([out], file.name, { type: 'image/gif' });
}
async function prepareImage(file) {
  if (file.type === 'image/gif') file = await gifLoop(file);
  const bmp = await createImageBitmap(file).catch(() => null);
  if (!bmp) throw new Error('Não consegui abrir a imagem "' + file.name + '".');
  const w = bmp.width, h = bmp.height, scale = Math.min(1, 1800 / Math.max(w, h));
  const small = file.type === 'image/gif' || (file.type === 'image/png' && file.size < 1.5e6) || (scale === 1 && file.size < 1.2e6);
  if (small) return { blob: file, type: file.type, ratio: nearestRatio(w, h) };
  const c = document.createElement('canvas'); c.width = Math.round(w * scale); c.height = Math.round(h * scale);
  const ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, c.width, c.height); ctx.drawImage(bmp, 0, 0, c.width, c.height);
  const blob = await new Promise(r => c.toBlob(r, 'image/jpeg', .86));
  return { blob, type: 'image/jpeg', ratio: nearestRatio(w, h) };
}
function videoRatio(file) {

function xhrPut(url, blob, onProg, reg) {
  return new Promise((res, rej) => {
    const x = new XMLHttpRequest(); if (reg) reg(x); x.onabort = () => rej(Object.assign(new Error('Envio cancelado.'), { cancelled: true })); x.open('PUT', url); x.withCredentials = true; x.setRequestHeader('x-requested-with', 'portfolio');
    x.upload.onprogress = e => e.lengthComputable && onProg(e.loaded);
    x.onload = () => { let d = {}; try { d = JSON.parse(x.responseText); } catch (e) {} x.status < 300 ? res(d) : rej(new ApiError(d.error || 'Falha no envio', x.status)); };
    x.onerror = () => rej(new Error('Falha de conexão durante o envio.'));
    x.send(blob);
  });
}
/* envia um arquivo/blob em pedaços e devolve o endereço salvo */
let uploading = 0;
addEventListener('beforeunload', e => { if (uploading) { e.preventDefault(); e.returnValue = ''; } });
async function sendBlob(blob, name, type, label) {
  let cancelled = false, xhr = null, id = null, cancelRej = null, initP = null;
  const stop = () => Object.assign(new Error('Envio cancelado.'), { cancelled: true });
  const cancelP = new Promise((_, rej) => { cancelRej = rej; }); cancelP.catch(() => {});
  const p = progress(label || 'Enviando ' + name + '…', () => { cancelled = true; try { if (xhr) xhr.abort(); } catch (e) {} cancelRej(stop()); }); uploading++;
  try {
    initP = api('POST', '/api/media/init', { name, type, size: blob.size }); initP.catch(() => {});
    const init = await Promise.race([initP, cancelP]); id = init.id;
    let sent = 0;
    for (let i = 0; i < init.chunks; i++) {
      if (cancelled) throw stop();
      const part = blob.slice(i * init.chunk, (i + 1) * init.chunk); const base = sent;
      await Promise.race([cancelP, xhrPut(`/api/media/${init.id}/${i}`, part, loaded => p.set((base + loaded) / blob.size, `${label || 'Enviando ' + name} — ${Math.round((base + loaded) / blob.size * 100)}%`), x => { xhr = x; })]);
      sent += part.size;
    }
    if (cancelled) throw stop();
    p.lock();
    const done = await api('POST', `/api/media/${init.id}/done`, {});
    p.done(); usage(); return done.src;
  } catch (e) {
    if (cancelled || e.cancelled) {
      e.cancelled = true; e.shown = true; p.done();
      if (id) { try { await api('DELETE', '/api/media/' + id); } catch (x) {} }
      else if (initP) initP.then(i => api('DELETE', '/api/media/' + i.id)).then(usage).catch(() => {});
      toast('Envio cancelado. O arquivo parcial foi apagado.', '', 3500); usage();
    } else { p.fail(e.message); e.shown = true; }
    throw e;
  }
  finally { uploading--; }
}
/* arquivo escolhido -> { src, t, r }. "pre" = imagem já recortada { blob, type, ratio } */
async function upload(file, pre) {
  const isVideo = !pre && (file.type.startsWith('video/') || /\.(mp4|mov|m4v|webm)$/i.test(file.name));
  try {
    let blob = file, type = file.type, ratio;
    if (pre) ({ blob, type, ratio } = pre);
    else if (isVideo) {
      if (!/^video\/(mp4|webm|quicktime)$/.test(type)) type = /\.webm$/i.test(file.name) ? 'video/webm' : 'video/mp4';
      if (type === 'video/quicktime') toast('Arquivos .MOV podem não abrir no Chrome/Windows. O ideal é enviar em MP4.', '', 7000);
      ratio = await videoRatio(file);
    } else {
      if (!file.type.startsWith('image/')) throw new Error('"' + file.name + '" não é imagem nem vídeo.');
      ({ blob, type, ratio } = await prepareImage(file));
    }
    const src = await sendBlob(blob, file.name, type);
    return { src, t: isVideo ? 'video' : 'design', r: ratio };
  } catch (e) { if (!e.shown) toast(e.message, 'bad', 6000); throw e; }
}

function renderCrop(img, sx, sy, sw, sh, maxSide) {
  const k = Math.min(1, maxSide / Math.max(sw, sh));
  const c = document.createElement('canvas'); c.width = Math.max(1, Math.round(sw * k)); c.height = Math.max(1, Math.round(sh * k));
  const ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, c.width, c.height);
  ctx.imageSmoothingQuality = 'high'; ctx.drawImage(img, sx, sy, sw, sh, 0, 0, c.width, c.height);
  return c;
}
const canvasBlob = (c, q = .88) => new Promise(r => c.toBlob(r, 'image/jpeg', q));
/* recorte central automático (cobre o quadro inteiro) */
async function autoCrop(source, ratio) {
  const url = typeof source === 'string' ? source : URL.createObjectURL(source);
  const img = await loadImg(url); if (typeof source !== 'string') URL.revokeObjectURL(url);
  const q = ratioNum(ratio); let sw = img.naturalWidth, sh = sw / q;
  if (sh > img.naturalHeight) { sh = img.naturalHeight; sw = sh * q; }
  const c = renderCrop(img, (img.naturalWidth - sw) / 2, (img.naturalHeight - sh) / 2, sw, sh, 1600);
  return { blob: await canvasBlob(c), type: 'image/jpeg', ratio };
}

      vwrap.innerHTML = '';

        v.append(pv, nx, dots);

      const source = s.file || s.src;
      const r = await cropDialog({ source: s.blob ? s.url : source, ratio, fixed: true, kind: 'carousel', title: 'Recortar foto do carrossel' }); if (!r) return;
      s.blob = r; s.url = URL.createObjectURL(r.blob); s.recrop = false; draw();

        for (const s of slides) {
          if (!need(s)) { images.push(s.src); continue; }
          n++; const label = `Enviando foto ${n} de ${total}`;
          if (s.gif && s.file && !s.blob) { images.push(await sendBlob(await gifLoop(s.file), s.file.name, 'image/gif', label)); continue; }

        }
        done({ title: title.value.trim(), ratio, images });
      } catch (ex) { err.textContent = ex.message; ok.disabled = false; ok.textContent = 'Salvar carrossel'; }
    };
    draw();
  });
}

function normHref(v, icon) {
  v = (v || '').trim(); if (!v) return '';
  if (/^(https?:|mailto:|tel:)/i.test(v)) return v;
  if (/^[\w.+-]+@[\w-]+(\.[\w-]+)+$/.test(v)) return 'mailto:' + v;
  if (icon === 'whatsapp' && /^[\d\s()+-]{8,}$/.test(v)) return 'https://wa.me/' + v.replace(/\D/g, '');
  if (icon === 'phone' && /^[\d\s()+-]{6,}$/.test(v)) return 'tel:' + v.replace(/[^\d+]/g, '');
  if (icon === 'instagram' && /^@?[A-Za-z0-9._]+$/.test(v)) return 'https://www.instagram.com/' + v.replace(/^@/, '') + '/';
  return 'https://' + v.replace(/^\/+/, '');
}

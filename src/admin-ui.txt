function toast(msg, kind, ms = 3500) {
  const t = document.createElement('div'); t.className = 'pf-toast ' + (kind || ''); t.textContent = msg; toasts.append(t);

function progress(label) {
  const t = document.createElement('div'); t.className = 'pf-toast';
  t.innerHTML = '<div></div><div class="pbar"><b></b></div>'; t.firstChild.textContent = label; toasts.append(t);
  return {
    set(p, txt) { $('b', t).style.width = Math.round(p * 100) + '%'; t.firstChild.textContent = txt || label; },
    done() { t.remove(); },
    fail(msg) { t.className = 'pf-toast bad'; t.textContent = msg; setTimeout(() => t.remove(), 7000); }
  };
}

class ApiError extends Error { constructor(m, status, data) { super(m); this.status = status; this.data = data; } }
async function api(method, path, body) {
  const r = await fetch(path, {
    method, credentials: 'same-origin',
    headers: { 'x-requested-with': 'portfolio', ...(body !== undefined ? { 'content-type': 'application/json' } : {}) },
    body: body !== undefined ? JSON.stringify(body) : undefined
  });
  const data = await r.json().catch(() => ({}));
  if (r.status === 401) { toast('Sua sessão acabou. Entre de novo.', 'bad', 0); setTimeout(() => location.href = '/admin', 1500); }
  if (!r.ok) throw new ApiError(data.error || 'Erro no servidor', r.status, data);
  return data;
}
let queue = Promise.resolve();
function save() {
  queue = queue.then(async () => {
    try {
      const r = await api('PUT', '/api/content', { clients: PF.clients, rev: PF.rev });
      PF.rev = r.rev; toast('Salvo ✓', 'ok', 1800);

      if (e.status === 409) { toast(e.message, 'bad', 0); setTimeout(() => location.reload(), 2500); }

    }
  });
  return queue.catch(() => {});
}

const RATIOS = [['9/16', 'Vertical (9:16)'], ['16/9', 'Horizontal (16:9)'], ['4/5', 'Retrato (4:5)'], ['1/1', 'Quadrado (1:1)'], ['4/3', 'Horizontal clássico (4:3)'], ['3/4', 'Retrato (3:4)']];
function nearestRatio(w, h) {
  const q = w / h; let best = '1/1', d = 9;
  for (const [r] of RATIOS) { const [a, b] = r.split('/').map(Number); const x = Math.abs(Math.log(q / (a / b))); if (x < d) { d = x; best = r; } }
  return best;
}
async function prepareImage(file) {
  const bmp = await createImageBitmap(file).catch(() => null);
  if (!bmp) throw new Error('Não consegui abrir a imagem "' + file.name + '".');
  const w = bmp.width, h = bmp.height, scale = Math.min(1, 1800 / Math.max(w, h));
  const small = file.type === 'image/gif' || (file.type === 'image/png' && file.size < 1.5e6) || (scale === 1 && file.size < 1.2e6);
  if (small) return { blob: file, type: file.type, ratio: nearestRatio(w, h) };
  const c = document.createElement('canvas'); c.width = Math.round(w * scale); c.height = Math.round(h * scale);
  const ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, c.width, c.height); ctx.drawImage(bmp, 0, 0, c.width, c.height);
  const blob = await new Promise(r => c.toBlob(r, 'image/jpeg', .86));
  return { blob, type: 'image/jpeg', ratio: nearestRatio(w, h) };
}
function videoRatio(file) {
  return new Promise(resolve => {
    const v = document.createElement('video'); v.preload = 'metadata'; v.muted = true; const u = URL.createObjectURL(file);
    v.onloadedmetadata = () => { const r = v.videoWidth ? nearestRatio(v.videoWidth, v.videoHeight) : '16/9'; URL.revokeObjectURL(u); resolve(r); };
    v.onerror = () => { URL.revokeObjectURL(u); resolve('16/9'); };
    v.src = u;
  });
}
function xhrPut(url, blob, onProg) {
  return new Promise((res, rej) => {
    const x = new XMLHttpRequest(); x.open('PUT', url); x.withCredentials = true; x.setRequestHeader('x-requested-with', 'portfolio');
    x.upload.onprogress = e => e.lengthComputable && onProg(e.loaded);
    x.onload = () => { let d = {}; try { d = JSON.parse(x.responseText); } catch (e) {} x.status < 300 ? res(d) : rej(new ApiError(d.error || 'Falha no envio', x.status)); };
    x.onerror = () => rej(new Error('Falha de conexão durante o envio.'));
    x.send(blob);
  });
}
/* envia um arquivo em pedaços e devolve { src, t, r } */
async function upload(file) {
  const isVideo = file.type.startsWith('video/') || /\.(mp4|mov|m4v|webm)$/i.test(file.name);
  const p = progress('Preparando ' + file.name + '…');
  try {
    let blob = file, type = file.type, ratio;
    if (isVideo) {
      if (!/^video\/(mp4|webm|quicktime)$/.test(type)) type = /\.webm$/i.test(file.name) ? 'video/webm' : 'video/mp4';
      if (type === 'video/quicktime') toast('Arquivos .MOV podem não abrir no Chrome/Windows. O ideal é enviar em MP4.', '', 7000);
      ratio = await videoRatio(file);
    } else {
      if (!file.type.startsWith('image/')) throw new Error('"' + file.name + '" não é imagem nem vídeo.');
      ({ blob, type, ratio } = await prepareImage(file));
    }
    const init = await api('POST', '/api/media/init', { name: file.name, type, size: blob.size });
    let sent = 0;
    for (let i = 0; i < init.chunks; i++) {
      const part = blob.slice(i * init.chunk, (i + 1) * init.chunk);
      const base = sent;
      await xhrPut(`/api/media/${init.id}/${i}`, part, loaded => p.set((base + loaded) / blob.size, `Enviando ${file.name} — ${Math.round((base + loaded) / blob.size * 100)}%`));
      sent += part.size;
    }
    const done = await api('POST', `/api/media/${init.id}/done`, {});
    p.done(); usage();
    return { src: done.src, t: isVideo ? 'video' : 'design', r: ratio };
  } catch (e) { p.fail(e.message); throw e; }
}

  });
}

function uniqueId(name) {

  const base = slug(name); let id = base, n = 2;
  while (used.has(id)) id = base + '-' + n++;
  return id;
}

  async 'replace-work'(id, i) {
    const it = findItem(id); const w = it && it.works[i]; if (!w) return;
    const [f] = await pickFiles('image/*,video/mp4,video/webm,video/quicktime,.mp4,.mov,.m4v,.webm', false); if (!f) return;
    try {
      const up = await upload(f); const old = mediaOf(w.src);
      w.src = up.src; w.t = up.t; w.r = up.r; delete w.canva;
      PF.refresh(); await save(); dropMedia(old);
    } catch (e) {}
  },
  async 'del-work'(id, i) {
    const it = findItem(id); const w = it && it.works[i]; if (!w) return;
    if (!await confirmBox('Excluir "' + w.title + '"?', 'O arquivo enviado também será apagado. Não dá para desfazer.')) return;
    const old = mediaOf(w.src); it.works.splice(i, 1);
    PF.refresh(); await save(); dropMedia(old);
  },
  async password() {
    const v = await modal({ title: 'Trocar senha', ok: 'Trocar', fields: [
      { key: 'current', label: 'Senha atual', type: 'password', required: true },
      { key: 'next', label: 'Nova senha', hint: 'Mínimo de 10 caracteres.', type: 'password', required: true }] }); if (!v) return;
    try { await api('POST', '/api/password', v); toast('Senha trocada ✓', 'ok'); } catch (e) { toast(e.message, 'bad', 6000); }
  },
  async logout() { try { await api('POST', '/api/logout', {}); } catch (e) {} location.reload(); },
  preview() { const on = document.body.classList.toggle('pf-preview'); $('[data-act=preview]').textContent = on ? 'Voltar a editar' : 'Ver como visitante'; }
};

  e.preventDefault(); e.stopPropagation();
  const a = b.dataset.act, id = b.dataset.id, i = b.dataset.i !== undefined ? Number(b.dataset.i) : undefined;
  actions[a](id, i);
}, true);
